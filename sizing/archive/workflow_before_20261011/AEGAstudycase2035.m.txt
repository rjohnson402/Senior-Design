function [m,o] = AEGAstudycase2035(msn,c)
% Resolve the common two-cruise/six-high-lift configuration from live inputs.
if nargin<2, c=AEGAstudysettings2035(msn); end
m=msn; m.AR=c.reference_AR; m.TR=c.reference_taper;
m.enforce_land=c.enforce_landing;
m.prop_D_side_ft=[repmat(c.high_lift_diameter_ft,1,3) c.cruise_diameter_ft];
anchor=c.anchor_high_lift_diameter_ft;
inner=m.WFus/2+anchor/2+c.prop_gap_ft;
m.prop_high_lift_y_side_ft=inner+(0:2)*(anchor+c.prop_gap_ft);
m.prop_y_fraction_side=[.2 .4 .6 1];
o=struct('coupled_flight_phases',true,'cruise_disk_indices',[4 8], ...
    'enforce_battery_out',true,'weight_tolerance',.01, ...
    'wing_mass_factor',c.technology.wing_mass_factor, ...
    'retract_fraction',0,'retraction_policy','manual','cruise_deployment',1, ...
    'cruise_stall_speed_factor',1.2,'cruise_CL_limit',m.CLmax_clean/1.2^2, ...
    'mission_profile',struct('name',sprintf('Current %.0f-knot mission',m.Vkt), ...
    'route_speed_ktas',m.Vkt,'route_distance_nm',m.range,'reserve_speed_ktas',m.Vkt));
end
